cmake_minimum_required(VERSION 3.16)
project(go2_observer_cameras)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(go2_chase_camera SHARED src/chase_camera_system.cpp)
target_include_directories(go2_chase_camera PRIVATE include)
target_link_libraries(go2_chase_camera PRIVATE gz-sim::gz-sim gz-plugin::register)
install(TARGETS go2_chase_camera LIBRARY DESTINATION lib)
install(DIRECTORY models DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(camera_pose_test test/camera_pose_test.cpp)
  target_include_directories(camera_pose_test PRIVATE include)
  target_link_libraries(camera_pose_test PRIVATE gz-sim::gz-sim)
  add_test(NAME camera_pose COMMAND camera_pose_test)
endif()
install(DIRECTORY include/ DESTINATION include)
ament_package()
